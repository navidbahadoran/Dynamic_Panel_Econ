Run the Revision-8 Monte Carlo. --pooled-r2-target calibrates the simulated
DGP through c_xi; it is not an estimation or rank-selection tuning parameter.

options:
  -h, --help            show this help message and exit
  --config CONFIG       Optional TOML configuration; CLI values override it.

simulation:
  --dgp {1,2,3,4}
  --dgp-grid DGP_GRID
  --N PANEL_N
  --T PANEL_T
  --balanced-grid BALANCED_GRID
  --replications REPLICATIONS
  --burn-in BURN_IN
  --seed SEED

DGP:
  --rho-g RHO_G
  --mu-f-a MU_F_A
  --kappa-f-a KAPPA_F_A
  --mu-f-b MU_F_B
  --kappa-f-b KAPPA_F_B
  --rho-x RHO_X
  --delta-x DELTA_X
  --rho-fx RHO_FX
  --eta-x ETA_X
  --rho-s RHO_S

  --mu-lambda-b-1 MU_LAMBDA_B_1
  --mu-lambda-b-2 MU_LAMBDA_B_2
  --sigma-lambda-b SIGMA_LAMBDA_B

calibration:
  --pooled-r2-target POOLED_R2_TARGET
                        Requested pooled DGP R-squared used to calibrate c_xi
                        (baseline 0.65). For rank-stress r_B=0, c_xi=1 and
                        induced R-squared is reported instead.
  --calibration-draws CALIBRATION_DRAWS
  --calibration-seed CALIBRATION_SEED
  --calibration-tolerance CALIBRATION_TOLERANCE

rank:
  --rank-mode {fixed,selected}

  --rank-caps RANK_CAPS
  --true-ranks TRUE_RANKS
